% Automatic bolding of Natalie Davidson's name in the bibliography.
% \mkbibnamefamily receives the family-name text directly as its argument,
% so we compare that text directly rather than relying on \namepartfamily
% (which isn't reliably set at this hook -- that was the earlier bug).
\newcommand*{\cvboldfamily}[1]{%
  \IfStrEq{#1}{Davidson}{\textbf{#1}}{#1}%
}
\renewcommand*{\mkbibnamefamily}[1]{\cvboldfamily{#1}}
