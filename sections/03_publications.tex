%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Publications
%%
%% \nocite{*} marks every entry in publications.bib as "cited" so it's
%% eligible to be printed. \printbibliography[keyword=mypub] then filters
%% that down to only entries tagged keywords={mypub} -- i.e. papers you
%% are actually an author on (including as a named consortium member).
%%
%% To add a new publication: add the BibTeX entry with keywords={mypub}
%% (or append ",mypub" to an existing keywords field). To cite a paper
%% elsewhere (e.g. in Software) WITHOUT it appearing in this list, leave
%% off the mypub tag.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\cvsection{Publications}

\nocite{*}

\printbibliography[
    heading=none,
    keyword=mypub
]
